115 public boolean movePlayer(int deltaX, int deltaY) {
116 if (isGameOver() || lm == null || player == null) return false;
117
118 boolean moved = turnSystem.executePlayerTurn(deltaX, deltaY);
119 for (GameEvent event : turnSystem.getPendingEvents()) {
120
121 if (event instanceof SystemEvent systemEvent) {
122 String message = systemEvent.toLogMessage();
123
124 if (message.startsWith("GAME_OVER:")) {
125 handlePlayerDeath();
126 continue;
127 }
128
129 if (message.startsWith("LEVEL_CLEARED:")) {
130 String levelStr = message.substring("LEVEL_CLEARED:".length());
131 int levelNumber;
132 try {
133 levelNumber = Integer.parseInt(levelStr);
134 } catch (NumberFormatException e) {
135 levelNumber = lm != null ? lm.getGameState().numberOfLevel() : 1;
136 }
137 handleLevelCleared(levelNumber);
138 continue;
139 }
140 }
141
142 addLogMessage(event.toLogMessage());
143 }
144 turnSystem.clearPendingEvents();
145
146 if (moved) {
147 player.timeHasPassed();
148 if (!gameOver) {
149 this.lastGameState = lm.getGameState();
150 }
151 }
152 return moved;
153 }
