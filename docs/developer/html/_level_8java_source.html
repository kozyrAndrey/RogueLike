91 public List<Corridor> generateCorridors(int[][] adjMatrix, ArrayList<Room> rooms, Random rng) {
92 ArrayList<Corridor> corridors = new ArrayList<>();
93 boolean[] visited = new boolean[9];
94 Stack<Integer> stack = new Stack<>();
95 stack.push(rng.nextInt(9));
96 visited[stack.peek()] = true;
97
98 while (!stack.isEmpty()) {
99 int current = stack.pop();
100 ArrayList<Integer> neighbors = new ArrayList<>();
101 for (int neighbor = 0; neighbor < 9; neighbor++) {
102 if (adjMatrix[current][neighbor] == 1 && !visited[neighbor])
103 neighbors.add(neighbor);
104 }
105 Collections.shuffle(neighbors, rng);
106 for (int neighbor : neighbors) {
107 if (!visited[neighbor]) {
108 visited[neighbor] = true;
109 stack.push(neighbor);
110 createCorridor(rooms, neighbor, current, rng, corridors);
111 }
112 }
113 }
114 int extraCorr = rng.nextInt(4);
115 for (int i = 0; i < extraCorr; i++) {
116 int room1 = rng.nextInt(9);
117 int room2 = rng.nextInt(9);
118
119 if (room1 != room2 && adjMatrix[room1][room2] == 1) {
120 boolean corrExists = false;
121 for (Corridor existing : corridors) {
122 if ((existing.getFirst() == room1 && existing.getSecond() == room2) ||
123 (existing.getFirst() == room2 && existing.getSecond() == room1)) {

126 }

242 private int[] calcDistances(ArrayList<ArrayList<Integer>> graph, int startRoom) {
243 int[] distances = new int[graph.size()];
244 Arrays.fill(distances, -1);
245 distances[startRoom] = 0;
246
247 Queue<Integer> queue = new ArrayDeque<>();
248 queue.add(startRoom);
249
250 while (!queue.isEmpty()) {
251 int current = queue.poll();
252 for (int neighbor : graph.get(current)) {
253 if (distances[neighbor] == -1) {
254 distances[neighbor] = distances[current] + 1;
255 queue.add(neighbor);
256 }
257 }
258 }
259
260 return distances;
