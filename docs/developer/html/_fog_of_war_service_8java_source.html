18 // Получить границы комнаты, если игрок в ней
19 private RoomBounds getRoomBounds(Coordinates playerPos, int[][] map) {
20 int height = map.length;
21 int width = map[0].length;
22
23 // Если игрок не в комнате, возвращаем null
24 if (!isPlayerInRoom(playerPos, map)) {
25 return null;
26 }
27
28 // Ищем границы комнаты с помощью flood fill
29 boolean[][] visited = new boolean[height][width];
30 Queue<Coordinates> queue = new LinkedList<>();
31 queue.add(playerPos);
32 visited[playerPos.y()][playerPos.x()] = true;
33
34 int minX = playerPos.x();
35 int maxX = playerPos.x();
36 int minY = playerPos.y();
37 int maxY = playerPos.y();
38
39 int[][] directions = {{0, -1}, {1, 0}, {0, 1}, {-1, 0}};
40
41 while (!queue.isEmpty()) {
42 Coordinates current = queue.poll();
43 int x = current.x();
44 int y = current.y();
45
46 // Обновляем границы
47 minX = Math.min(minX, x);
48 maxX = Math.max(maxX, x);
49 minY = Math.min(minY, y);
50 maxY = Math.max(maxY, y);
51
52 // Проверяем соседей
53 for (int[] dir : directions) {
54 int nx = x + dir[0];
55 int ny = y + dir[1];
56
57 if (nx >= 0 && nx < width && ny >= 0 && ny < height
58 && !visited[ny][nx] && map[ny][nx] == 1) {
59 visited[ny][nx] = true;
60 queue.add(new Coordinates(ny, nx));
61 }
62 }
63 }
64
65 return new RoomBounds(minX, maxX, minY, maxY);
66 }

81 private List<Coordinates> bresenhamLine(int x0, int y0, int x1, int y1) {
82 List<Coordinates> line = new ArrayList<>();
83
84 int dx = Math.abs(x1 - x0);
85 int dy = Math.abs(y1 - y0);
86
87 int sx = x0 < x1 ? 1 : -1;
88 int sy = y0 < y1 ? 1 : -1;
89
90 int err = dx - dy;
91
92 while (true) {
93 line.add(new Coordinates(y0, x0));
94
95 if (x0 == x1 && y0 == y1) break;
96
97 int e2 = 2 * err;
98 if (e2 > -dy) {
99 err -= dy;
100 x0 += sx;
101 }
102 if (e2 < dx) {
103 err += dx;
104 y0 += sy;
105 }
106 }
107
108 return line;
109 }

116 // Ray Casting с алгоритмом Брезенхэма
117 private void bresenhamRayCasting(int centerX, int centerY, int height, int width,
118 FogState[][] newVisibility, int[][] map) {
119 for (int y = Math.max(0, centerY - visionRadius);
120 y <= Math.min(height - 1, centerY + visionRadius); y++) {
121 for (int x = Math.max(0, centerX - visionRadius);
122 x <= Math.min(width - 1, centerX + visionRadius); x++) {
123
124 double distance = Math.sqrt(Math.pow(x - centerX, 2) + Math.pow(y - centerY, 2));
125 if (distance > visionRadius) continue;
126
127 List<Coordinates> line = getBresenhamLineCached(centerX, centerY, x, y);
128 boolean blocked = false;
129
130 for (Coordinates point : line) {
131 int px = point.x();
132 int py = point.y();
133
134 if (px < 0 || px >= width || py < 0 || py >= height) {
135 blocked = true;
136 break;
137 }
138
139 if (px == x && py == y) {
140 if (!blocked) newVisibility[py][px] = FogState.VISIBLE;
141 break;
142 }
143
144 // Показываем текущую клетку
145 newVisibility[py][px] = FogState.VISIBLE;
146
147 int cell = map[py][px];
148
149 // Проверяем, не блокирует ли стена, закрытая дверь, пустота
150 if (cell == 3 || (cell >= 4 && cell <= 6) || cell== 0) {
151 blocked = true;
152 break;
153 }
154 }
155 }
156 }

159 private void revealCorridor(int centerX, int centerY, int[][] map, FogState[][] fog) {
160 int height = map.length;
161 int width = map[0].length;
162
163 // Основные направления коридоров
164 int[][] directions = {{0, -1}, {1, 0}, {0, 1}, {-1, 0}};
165
166 for (int[] dir : directions) {
167 int dx = dir[0];
168 int dy = dir[1];
169 int x = centerX + dx;
170 int y = centerY + dy;
171
172 // Идем вдоль коридора в каждом направлении
173 while (x >= 0 && x < width && y >= 0 && y < height) {
174 int cell = map[y][x];
175
176 // Прекращаем если это не коридор
177 if (cell != 2) {
178 // Но показываем граничную клетку (дверь или стену)
179 if (cell == 3 || (cell >= 4 && cell <= 6)) {
180 fog[y][x] = FogState.VISIBLE;
181 }
182 break;
183 }
184
185 fog[y][x] = FogState.VISIBLE;
186
187 x += dx;
188 y += dy;
189 }
190 }

193 public FogState[][] calculateVisibility(Coordinates playerPos, int[][] map) {
194 int height = map.length;
195 int width = map[0].length;
196 FogState[][] newFog = new FogState[height][width];
197
198 // Инициализируем все как HIDDEN
199 for (int y = 0; y < height; y++) {
200 Arrays.fill(newFog[y], FogState.HIDDEN);
201 }
202
203 int centerX = playerPos.x();
204 int centerY = playerPos.y();
205
206 if (centerX < 0 || centerX >= width || centerY < 0 || centerY >= height) {
207 return newFog; // Игрок вне карты
208 }
209 // Проверяем, находится ли игрок в комнате
210 if (map[centerY][centerX] == 1) {
211 // Игрок в комнате - показываем ВСЮ комнату
212 RoomBounds bounds = getRoomBounds(playerPos, map);
213 if (bounds != null) {
214 for (int y = Math.max(0, bounds.minY - 1); y <= Math.min(height - 1, bounds.maxY + 1); y++) {
215 for (int x = Math.max(0, bounds.minX - 1); x <= Math.min(width - 1, bounds.maxX + 1); x++) {
216 // Пол комнаты и стены, и начало пола коридора
217 if (map[y][x] == 1 || map[y][x] == 3 || map[y][x] == 2) {
218 newFog[y][x] = FogState.VISIBLE;
219 }
220 }
221 }
222 }
223 } else if (map[centerY][centerX] == 2) {
224 // Игрок в коридоре - используем ray casting
225 newFog[centerY][centerX] = FogState.VISIBLE;
226
227 // 1. Ray Casting с алгоритмом Брезенхэма
228 bresenhamRayCasting(centerX, centerY, height, width, newFog, map);
229
230 // 2. Видимость вдоль коридоров (дополнительно)
231 revealCorridor(centerX, centerY, map, newFog);
232 }
233 return newFog;
234 }

239 public FogState[][] updateFog(FogState[][] oldFog, FogState[][] newVisibility) {
240 int height = oldFog.length;
241 int width = oldFog[0].length;
242 FogState[][] updatedFog = new FogState[height][width];
243
244 for (int y = 0; y < height; y++) {
245 for (int x = 0; x < width; x++) {
246 FogState oldState = oldFog[y][x];
247 FogState newState = newVisibility[y][x];
248
249 if (newState == FogState.VISIBLE) {
250 updatedFog[y][x] = FogState.VISIBLE;
251 } else if (oldState == FogState.VISIBLE || oldState == FogState.EXPLORED) {
252 updatedFog[y][x] = FogState.EXPLORED;
253 } else {
254 updatedFog[y][x] = FogState.HIDDEN;
255 }
256 }
257 }
258
259 return updatedFog;
