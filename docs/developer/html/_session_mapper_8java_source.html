111 private static ItemSnapshot toItemSnapshot(Item item) {
112 int value = 0;
113 String boostType = null;
114 String accessLevel = null;
115 if (item instanceof Weapon w) {
116 value = w.getPower();
117 } else if (item instanceof Food f) {
118 value = f.getValue();
119 } else if (item instanceof Scroll s) {
120 value = s.getValue();
121 boostType = s.getBoostType().name();
122 } else if (item instanceof Elixir e) {
123 value = e.getValue();
124 boostType = e.getBoostType().name();
125 } else if (item instanceof Treasure t) {
126 value = t.getValue();
127 } else if (item instanceof Key k) {
128 accessLevel = k.getAccess().name();
129 }
130 return new ItemSnapshot(item.getType().name(), item.getCoor().x(), item.getCoor().y(), value, boostType,
131 accessLevel, item.getName());
132 }

134 private static Item fromItemSnapshot(ItemSnapshot s) {
135 Coordinates coor = new Coordinates(s.y(), s.x());
136 return switch (s.type()) {
137 case "WEAPON" -> new Weapon(coor, s.name(), s.value());
138 case "FOOD" -> new Food(coor, s.name(), s.value());
139 case "SCROLL" -> {
140 BoostType bt = BoostType.valueOf(s.boostType());
141 yield new Scroll(coor, s.name(), bt, s.value());
142 }
143 case "ELIXIR" -> {
144 BoostType bt = BoostType.valueOf(s.boostType());
145 yield new Elixir(coor, s.name(), bt, s.value());
146 }
147 case "TREASURE" -> new Treasure(coor, s.value());
148 case "KEY" -> {
149 AccessLevel access = AccessLevel.valueOf(s.accessLevel());
150 yield new Key(coor, access, s.name());
151 }
152 default -> throw new IllegalArgumentException("Unknown item type");
153 };
154 }

225 public static LevelManager fromLevelSnapshot(LevelStateDTO s) {
226 Hero hero = fromHeroSnapshot(s.hero());
227 int prevDifficulty = s.balancePrevDifficulty();
228 int maxHpOnStart = s.balanceMaxHpOnStart();
229 int startValue = s.balanceStartValue();
230 validateBalanceData(prevDifficulty, maxHpOnStart, startValue);
231 BalanceService balanceService = new BalanceService(hero, prevDifficulty, maxHpOnStart, startValue);
232 LevelManager lm = new LevelManager(s.levelNumber(), hero, balanceService);
233 lm.addEntity(hero);
234 int[][] field = new int[s.field().length][];
235 for (int i = 0; i < s.field().length; i++) {
236 field[i] = s.field()[i].clone();
237 }
238 boolean[][] explored = new boolean[s.explored().length][];
239 for (int i = 0; i < s.explored().length; i++) {
240 explored[i] = s.explored()[i].clone();
241 }
242 lm.setField(field);
243 lm.setExplored(explored);
244 for (int i = 0; i < s.enemies().size(); i++) {
245 lm.addEntity(fromEnemySnapshot(s.enemies().get(i)));
246 }
247 for (int i = 0; i < s.items().size(); i++) {
248 lm.addEntity(fromItemSnapshot(s.items().get(i)));
249 }
250 for (int i = 0; i < s.doors().size(); i++) {
251 Door door = new Door(new Coordinates(s.doors().get(i).y(), s.doors().get(i).x()),
252 AccessLevel.valueOf(s.doors().get(i).accessLevel()));
253 lm.addEntity(door);
254 }
255 Exit exit = new Exit(new Coordinates(s.exit().y(), s.exit().x()));
256 lm.addEntity(exit);
257 return lm;
258 }
