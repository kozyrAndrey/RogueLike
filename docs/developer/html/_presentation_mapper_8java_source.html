29 private FieldPresentation mapField(int[][] field, FogState[][] fog) {
30 int height = field.length;
31 int width = field[0].length;
32
33 char[][] symbols = new char[height][width];
34 TextColor[][] colors = new TextColor[height][width];
35
36 for (int i = 0; i < height; i++) {
37 for (int j = 0; j < width; j++) {
38 int cell = field[i][j];
39 FogState state = fog[i][j];
40 switch (state) {
41 case HIDDEN:
42 // НЕИССЛЕДОВАННОЕ: АБСОЛЮТНО НИЧЕГО не показываем
43 symbols[i][j] = ' ';
44 colors[i][j] = TextColor.ANSI.BLACK;
45 break;
46
47 case EXPLORED:
48 // ИССЛЕДОВАННОЕ: показываем только пол, стены, двери
49 // НЕ показываем врагов, предметы, игрока
50 symbols[i][j] = getExploredSymbol(cell);
51 colors[i][j] = getExploredColor(cell);
52 break;
53
54 case VISIBLE:
55 // ВИДИМОЕ: показываем всё
56 symbols[i][j] = getSymbolForCell(cell);
57 colors[i][j] = getColorForCell(cell);
58 break;
59 }
60 }
61 }
62
63 return new FieldPresentation(symbols, colors, width, height);
64 }

195 private String formatHealthBar(int currentHp, int maxHp) {
196 float healthPercentage = (float) currentHp / maxHp;
197 int filled = (int) (10 * healthPercentage);
198 int empty = 10 - filled;
199 StringBuilder bar = new StringBuilder();
200
201 bar.append("HP[");
202 for (int i = 0; i < filled; i++) bar.append('█');
203 for (int i = 0; i < empty; i++) bar.append('░');
204 bar.append(String.format("] %3d/%3d ", currentHp, maxHp));
205
206 return bar.toString();
207 }
