79 private void handleMenuSelection() {
80 MenuItem selected = menuManager.getSelectedItem();
81 if (selected != null && selected.isEnabled()) {
82 switch (menuManager.getSelectedIndex()) {
83 case NEW_GAME_INDEX -> {
84 appState.setState(AppState.NAME_INPUT);
85 playerSession.reset();
86 uiManager.reset();
87 }
88 case CONTINUE_INDEX -> {
89 if (model.hasSavedSession() && model.loadLastSession()) {
90 appState.setState(AppState.GAME);
91 uiManager.setNormalState();
92 } else {
93 System.out.println("Unable to load save");
94 updateContinueMenuItem();
95 }
96 }
97 case STATISTICS_INDEX -> {
98 appState.setState(AppState.STATS);
99 uiManager.setNormalState();
100 }
101 }
102 }
103 }

129 private void handleGameCommand(Command command) {
130 if (uiManager.isShowingItemSelection()) {
131 UIState uiState = uiManager.getCurrentState();
132 if (UIState.BACKPACK_ROOT.equals(uiState.selectionType())) {
133 handleBackpackRootCommand(command);
134 } else {
135 handleItemSelectionCommand(command, uiState);
136 }
137 return;
138 }
139
140 switch (command.getType()) {
141 case MOVE_UP -> model.movePlayer(0, -1);
142 case MOVE_RIGHT -> model.movePlayer(1, 0);
143 case MOVE_DOWN -> model.movePlayer(0, 1);
144 case MOVE_LEFT -> model.movePlayer(-1, 0);
145 case OPEN_BACKPACK_MENU -> openBackpackRoot();
146 case QUIT_GAME -> {
147 working = false;
148 model.saveCurrentSession();
149 }
150 case MENU_BACK -> {
151 model.saveCurrentSession();
152 appState.setState(AppState.START_MENU);
153 uiManager.setNormalState();
154 updateContinueMenuItem();
155 }
156 default -> {}
157 }
158 if (isGameOver()) {
159 appState.setState(AppState.GAME_OVER);
160 uiManager.setNormalState();
161 }
162 }

208 private void handleItemSelectionCommand(Command command, UIState uiState) {
209 switch (command.getType()) {

230 private void useSelectedItem(int index, UIState uiState) {
231 Item selectedItem = model.getItemByIndex(index - 1);
232 if (selectedItem != null) {
233 String selectionType = uiState.selectionType();
234 ActionResult result = switch (selectionType) {
235 case "FOOD" -> model.useFood(selectedItem);
236 case "ELIXIR" -> model.useElixir(selectedItem);
237 case "SCROLL" -> model.useScroll(selectedItem);
238 case "WEAPON" -> model.equipWeapon(selectedItem);
239 default -> ActionResult.failure();
240 };
241 model.processActionResult(result);
242 }
243 uiManager.closeItemSelection();
244 }
